library(moments)

jb.daily <- JBTest(X)

strongest.stock <- names(
  which.min(jb.daily)
)

strongest.returns <- X[, strongest.stock]

qqnorm(
  strongest.returns,
  main = paste(
    "Normal QQ-plot:",
    strongest.stock
  )
)

qqline(
  strongest.returns,
  col = "red",
  lwd = 2
)

stock.skewness <- skewness(
  strongest.returns
)

stock.kurtosis <- kurtosis(
  strongest.returns
)

rejection.driver <- if (
  abs(stock.skewness) > 1 &&
  stock.kurtosis > 3
) {
  "Both skewness and fat tails"
} else if (
  stock.kurtosis > 3
) {
  "Fat tails"
} else {
  "Skewness"
}

c(
  stock = strongest.stock,
  jb_p_value = jb.daily[strongest.stock],
  skewness = stock.skewness,
  kurtosis = stock.kurtosis,
  driver = rejection.driver
)
