library(quantmod)

url <- "https://fred.stlouisfed.org/graph/fredgraph.csv?id=DCOILWTICO"

fred <- read.csv(
  url,
  na.strings = "."
)

fred$observation_date <- as.Date(fred$observation_date)
fred <- na.omit(fred)

wti <- xts(
  fred$DCOILWTICO,
  order.by = fred$observation_date
)["2025-06-01/"]
