run_study <- function(n_sims, capital, target, p) {
  ruined <- logical(n_sims)
  duration <- numeric(n_sims)

  for (s in 1:n_sims) {
    result <- simulate_trader(capital, target, p)
    ruined[s] <- result$ruined
    duration[s] <- result$days
  }

  c(prob_ruin = mean(ruined), avg_days = mean(duration))
}

set.seed(202607)

n_sims <- 2000

rbind(
  `p = 0.50` = run_study(n_sims, capital = 100, target = 200, p = 0.50),
  `p = 0.49` = run_study(n_sims, capital = 100, target = 200, p = 0.49),
  `p = 0.10` = run_study(n_sims, capital = 100, target = 200, p = 0.10)
)

rbind(
  `k = 50` = run_study(n_sims, capital = 50, target = 200, p = 0.50),
  `k = 100` = run_study(n_sims, capital = 100, target = 200, p = 0.50),
  `k = 150` = run_study(n_sims, capital = 150, target = 200, p = 0.50)
)

rbind(
  `N = 150` = run_study(n_sims, capital = 100, target = 150, p = 0.50),
  `N = 200` = run_study(n_sims, capital = 100, target = 200, p = 0.50),
  `N = 400` = run_study(n_sims, capital = 100, target = 400, p = 0.50)
)
